def get_db_connection():
    conn = sqlite3.connect('family.db')

# Функция-помощник для реверса связей
def get_reverse_role(role):
    if role == 'parent':
        return 'child'
    elif role == 'child':
        return 'parent'
    elif role == 'spouse':
        return 'spouse'
    elif role == 'sibling':
        return 'sibling'
    return role

    conn.close()

    for row in rows:
        person = dict(row)

        person['connections'] = json.loads(person['connections']) if person['connections'] else {}
        family[person['id']] = person
    return family

@app.route('/edit/<person_id>', methods=['GET', 'POST'])
def edit_relative(person_id):
    if not g.user or g.user['id'] != person_id:
        flash('У вас нет прав для редактирования этого профиля', 'error')
        return redirect(url_for('profile', person_id=person_id))
    conn = get_db_connection()
    row = conn.execute('SELECT * FROM relatives WHERE id = ? AND is_deleted = 0', (person_id,)).fetchone()
    
    if not row:
        conn.close()
        return "Родственник не найден", 404
        
    person = dict(row)
    # Распаковываем текущие связи из текста в словарь
    person['connections'] = json.loads(person['connections']) if person['connections'] else {}
        
    if request.method == 'POST':
        photo_file = request.files.get('photo')
        photo_filename = person['photo']
        
        if photo_file and photo_file.filename != '':
            photo_filename = photo_file.filename
            photo_file.save(os.path.join('static', photo_filename))
            
        # 1. ОБНОВЛЯЕМ СУЩЕСТВУЮЩИЕ СВЯЗИ ИЗ ФОРМЫ
        connections = person['connections']
        for conn_id in list(connections.keys()):
            field_name = f'edit_conn_{conn_id}'
            if field_name in request.form:
                new_val = request.form.get(field_name).strip()
                if new_val:
                    connections[conn_id] = new_val
                else:
                    del connections[conn_id]

        # 2. ЛОВИМ НОВЫЕ СВЯЗИ ИЗ ФОРМЫ
        new_conn_ids = request.form.getlist('new_connection_id[]')
        new_conn_roles = request.form.getlist('new_connection_role[]')
        
        for conn_id, conn_role in zip(new_conn_ids, new_conn_roles):
            conn_id = conn_id.strip() if conn_id else ""
            conn_role = conn_role.strip() if conn_role else ""
            if conn_id and conn_role:
                connections[conn_id] = conn_role
                
                # Получаем текущие связи родственника B
                b_row = conn.execute('SELECT connections FROM relatives WHERE id = ?', (conn_id,)).fetchone()
                if b_row:
                    b_connections = json.loads(b_row['connections']) if b_row['connections'] else {}
                    # Добавляем реверсивную связь: A -> get_reverse_role(X)
                    b_connections[person_id] = get_reverse_role(conn_role)
                    # Записываем обратно в базу для B
                    conn.execute('UPDATE relatives SET connections = ? WHERE id = ?', (
                        json.dumps(b_connections, ensure_ascii=False),
                        conn_id
                    ))
            
        # 3. ЗАПАКОВЫВАЕМ СЛОВАРЬ ОБРАТНО В ТЕКСТ ДЛЯ SQL
        connections_json = json.dumps(connections, ensure_ascii=False)
            
        # SQL-команда обновления (добавили connections)
        conn.execute('''
            UPDATE relatives 
            SET name = ?, role = ?, years = ?, bio = ?, photo = ?, connections = ?
            WHERE id = ?
        ''', (
            request.form.get('name'),
            request.form.get('role'),
            request.form.get('years'),
            request.form.get('bio'),
            photo_filename,
            connections_json,
            person_id
        ))
        conn.commit()
        conn.close()
        return redirect(url_for('profile', person_id=person_id))
        
    # Для GET-запроса получаем список всех родственников для выпадающего списка
    all_family = get_all_active_family()
    # Убираем из списка самого себя, чтобы нельзя было сослаться на себя же
    if person_id in all_family:
        del all_family[person_id]
        
    conn.close()
    # Передаем all_family в шаблон
    return render_template('edit.html', person_id=person_id, person=person, all_family=all_family)

import sqlite3

def clear_database():
    conn = sqlite3.connect('family.db')
    cursor = conn.cursor()
    
    try:
        # Get list of tables
        cursor.execute("SELECT name FROM sqlite_master WHERE type='table';")
        tables = [t[0] for t in cursor.fetchall() if not t[0].startswith('sqlite_')]
        print(f"Found tables: {tables}")
        
        # Clear memories table first due to foreign key references if any (optional but good practice)
        if 'memories' in tables:
            cursor.execute("DELETE FROM memories;")
            print("Cleared table 'memories'.")
            
        # Clear relatives table
        if 'relatives' in tables:
            cursor.execute("DELETE FROM relatives;")
            print("Cleared table 'relatives'.")
            
        # Reset autoincrement counters
        cursor.execute("DELETE FROM sqlite_sequence WHERE name IN ('memories', 'relatives');")
        conn.commit()
        print("Database family.db cleared successfully. All tables are empty, schemas preserved.")
    except Exception as e:
        conn.rollback()
        print(f"Error occurred: {e}")
    finally:
        conn.close()

if __name__ == '__main__':
    clear_database()

import sqlite3

def create_table():
    conn = sqlite3.connect('family.db')
    cursor = conn.cursor()
    
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS memories (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            target_person_id TEXT NOT NULL,
            author_id TEXT NOT NULL,
            life_stage TEXT NOT NULL,
            media_type TEXT NOT NULL,
            content TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY(target_person_id) REFERENCES relatives(id),
            FOREIGN KEY(author_id) REFERENCES relatives(id)
        )
    ''')
    
    conn.commit()
    conn.close()
    print("Table 'memories' created successfully in family.db.")

if __name__ == '__main__':
    create_table()

import os

def export_project():
    output_filename = 'project_snapshot.txt'
    
    # 1. Collect all files to read
    files_to_read = []
    
    # - Python files in the root directory
    for f in sorted(os.listdir('.')):
        if os.path.isfile(f) and f.endswith('.py'):
            files_to_read.append(f)
            
    # - HTML files inside templates/
    templates_dir = 'templates'
    if os.path.exists(templates_dir):
        for root, dirs, files in os.walk(templates_dir):
            dirs.sort()
            for file in sorted(files):
                if file.endswith('.html'):
                    rel_path = os.path.join(root, file)
                    rel_path = rel_path.replace('\\', '/')
                    files_to_read.append(rel_path)
                    
    # - CSS and JS files inside static/
    static_dir = 'static'
    if os.path.exists(static_dir):
        for root, dirs, files in os.walk(static_dir):
            dirs.sort()
            for file in sorted(files):
                if file.endswith('.css') or file.endswith('.js'):
                    rel_path = os.path.join(root, file)
                    rel_path = rel_path.replace('\\', '/')
                    files_to_read.append(rel_path)
                    
    # 2. Write to project_snapshot.txt
    with open(output_filename, 'w', encoding='utf-8') as outfile:
        for filepath in files_to_read:
            # Skip reading project_snapshot.txt itself if it somehow got in
            if os.path.basename(filepath) == output_filename:
                continue
                
            ext = os.path.splitext(filepath)[1].lstrip('.')
            
            try:
                with open(filepath, 'r', encoding='utf-8') as infile:
                    content = infile.read()
            except Exception as e:
                print(f"Error reading {filepath}: {e}")
                continue
                
            outfile.write(f"### Файл: `{filepath}`\n")
            outfile.write(f"```{ext}\n")
            outfile.write(content)
            if not content.endswith('\n'):
                outfile.write('\n')
            outfile.write("```\n\n")
            
    print(f"Done! Project snapshot successfully saved to: {output_filename}")

if __name__ == '__main__':
    export_project()
